cmake_minimum_required(VERSION 3.17)
project(reference_builders LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED)
find_package(ZLIB REQUIRED)
# Upstream checkouts (CGIF, PathSeer, FAVOR) at the commits in reference-revisions.json.
set(BASELINES "${CMAKE_CURRENT_SOURCE_DIR}/upstream" CACHE PATH "Directory holding the upstream checkouts")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../modules/baselines/include")
foreach(kind pathseer cgif)
  add_executable(${kind}_build build_index.cpp)
  target_compile_options(${kind}_build PRIVATE -march=native)
  target_link_libraries(${kind}_build PRIVATE OpenMP::OpenMP_CXX ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ZLIB::ZLIB)
endforeach()
target_compile_definitions(pathseer_build PRIVATE BUILD_PATHSEER)
target_include_directories(pathseer_build PRIVATE "${BASELINES}/PathSeer/faiss")
target_link_libraries(pathseer_build PRIVATE "${BASELINES}/PathSeer/faiss/build/faiss/libfaiss.a")
target_include_directories(cgif_build PRIVATE "${BASELINES}/CGIF/third_party/faiss")
target_link_libraries(cgif_build PRIVATE "${BASELINES}/CGIF/third_party/faiss/build/faiss/libfaiss.a")
add_executable(count_pathseer_edges count_pathseer_edges.cpp)
target_include_directories(count_pathseer_edges PRIVATE "${BASELINES}/PathSeer/faiss")
target_link_libraries(count_pathseer_edges PRIVATE "${BASELINES}/PathSeer/faiss/build/faiss/libfaiss.a" OpenMP::OpenMP_CXX ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ZLIB::ZLIB)
foreach(kind pathseer cgif)
  add_executable(${kind}_export_search export_search.cpp)
  target_link_libraries(${kind}_export_search PRIVATE OpenMP::OpenMP_CXX ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ZLIB::ZLIB)
endforeach()
target_compile_definitions(pathseer_export_search PRIVATE BUILD_PATHSEER)
target_include_directories(pathseer_export_search PRIVATE "${BASELINES}/PathSeer/faiss")
target_link_libraries(pathseer_export_search PRIVATE "${BASELINES}/PathSeer/faiss/build/faiss/libfaiss.a")
target_include_directories(cgif_export_search PRIVATE "${BASELINES}/CGIF/third_party/faiss")
target_link_libraries(cgif_export_search PRIVATE "${BASELINES}/CGIF/third_party/faiss/build/faiss/libfaiss.a")
add_executable(check_search_port check_search_port.cpp)
target_include_directories(check_search_port PRIVATE "${BASELINES}/PathSeer/faiss")
target_link_libraries(check_search_port PRIVATE "${BASELINES}/PathSeer/faiss/build/faiss/libfaiss.a" OpenMP::OpenMP_CXX ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ZLIB::ZLIB)
add_executable(check_favor_port check_favor_port.cpp)
set_property(TARGET check_favor_port PROPERTY CXX_STANDARD 20)
target_compile_options(check_favor_port PRIVATE -mavx2)
target_include_directories(check_favor_port PRIVATE "${BASELINES}/FAVOR/include")
add_executable(check_cgif_port check_cgif_port.cpp)
target_include_directories(check_cgif_port PRIVATE "${BASELINES}/CGIF/src" "${BASELINES}/CGIF/include" "${BASELINES}/CGIF/include/common")
target_sources(check_cgif_port PRIVATE "${BASELINES}/CGIF/src/common/utils.cc")
target_compile_options(check_cgif_port PRIVATE -ffunction-sections -fdata-sections)
target_link_options(check_cgif_port PRIVATE -Wl,--gc-sections)
